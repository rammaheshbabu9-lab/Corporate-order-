const SUPABASE_URL =
"https://eaowxilhfzyukzetgqxo.supabase.co";

const SUPABASE_KEY =
"sb_publishable_Z3xsADypaOjX8GEQEOLUdg_vbxynHRx";

let businesses = [];

async function supabaseRequest(path, options={}){

  const response =
    await fetch(
      `${SUPABASE_URL}/rest/v1/${path}`,
      {
        ...options,
        headers:{
          "apikey":SUPABASE_KEY,
          "Authorization":`Bearer ${SUPABASE_KEY}`,
          "Content-Type":"application/json",
          ...(options.headers || {})
        }
      }
    );

  const text =
    await response.text();

  if(!response.ok){
    throw new Error(
      text || `HTTP ${response.status}`
    );
  }

  return text ? JSON.parse(text) : null;
}

function showToast(message){

  const toast =
    document.getElementById("toast");

  toast.textContent = message;
  toast.classList.add("show");

  setTimeout(()=>{
    toast.classList.remove("show");
  },3500);
}

async function loadBusinesses(){

  try{

    businesses =
      await supabaseRequest(
        "businesses?select=*&order=created_at.desc"
      ) || [];

    renderTable();

    updateStats();

  }catch(error){

    console.error(error);

    document.getElementById("businessTable")
      .innerHTML =
      `<tr>
        <td colspan="11" class="loading">
          Failed to load businesses.
        </td>
      </tr>`;

    showToast("Database load failed.");
  }
}

function updateStats(){

  const total = businesses.length;

  const pending =
    businesses.filter(
      b => b.approval_status === "pending"
    ).length;

  const approved =
    businesses.filter(
      b => b.approval_status === "approved"
    ).length;

  const rejected =
    businesses.filter(
      b => b.approval_status === "rejected"
    ).length;

  const payment =
    businesses.filter(
      b => b.payment_status === "received"
    ).length;

  document.getElementById("totalCount")
    .textContent = total;

  document.getElementById("pendingCount")
    .textContent = pending;

  document.getElementById("approvedCount")
    .textContent = approved;

  document.getElementById("rejectedCount")
    .textContent = rejected;

  document.getElementById("paymentCount")
    .textContent = payment;
}

function renderTable(){

  const search =
    document.getElementById("search")
      .value.toLowerCase();

    document.getElementById("sectorFilter").value;

  const approval =
    document.getElementById("approvalFilter").value;

  const payment =
    document.getElementById("paymentFilter").value;

  const filtered =
    businesses.filter(b=>{

      const text =
        `${b.company_name || ""}
         ${b.mobile || ""}
         ${b.website || ""}
         ${b.contact_person || ""}`
         .toLowerCase();

      return
        (!search || text.includes(search)) &&
        (!sector || b.sector === sector) &&
        (!approval || b.approval_status === approval) &&
        (!payment || b.payment_status === payment);

    });

  const tbody =
    document.getElementById("businessTable");

  if(!filtered.length){

    tbody.innerHTML =
      `<tr>
        <td colspan="11" class="loading">
          No businesses found.
        </td>
      </tr>`;

    return;
  }

  tbody.innerHTML =
    filtered.map(b=>{

      const approvalClass =
        b.approval_status === "approved"
        ? "approved"
        : b.approval_status === "rejected"
        ? "rejected"
        : "pending";

      const paymentClass =
        b.payment_status === "received"
        ? "received"
        : b.payment_status === "not_received"
        ? "notreceived"
        : "pending";

      return `

<tr>

<td>
${
  b.logo_url
  ? `<img class="logo-img"
      src="${escapeAttr(b.logo_url)}">`
  : "—"
}
</td>

<td>
<strong>${escapeHtml(b.company_name)}</strong>
</td>

<td>
${escapeHtml(b.contact_person || "-")}
</td>

<td>
${escapeHtml(b.mobile || "-")}
</td>

<td>
${escapeHtml(b.sector || "Other")}
</td>

<td>
#${b.block_number || "-"}
</td>

<td>
₹${Number(b.price || 0).toFixed(0)}
</td>

<td>
<span class="status ${paymentClass}">
${escapeHtml(
  b.payment_status || "pending"
)}
</span>
</td>

<td>
<span class="status ${approvalClass}">
${escapeHtml(
  b.approval_status || "pending"
)}
</span>
</td>

<td>
${escapeHtml(
  b.booking_status || "available"
)}
</td>

<td>

<div class="actions">

<button
class="btn btn-gray"
data-action="view"
data-id="${b.id}">
View
</button>

<button
class="btn btn-gray"
data-action="edit"
data-id="${b.id}">
Edit
</button>

${
  b.payment_status !== "received"
  ?
  `<button
    class="btn btn-green"
    data-action="payment"
    data-id="${b.id}">
    Payment Received
  </button>`
  :
  ""
}

${
  b.approval_status !== "approved"
  ?
  `<button
    class="btn btn-green"
    data-action="approve"
    data-id="${b.id}">
    Approve
  </button>`
  :
  ""
}

${
  b.approval_status !== "rejected"
  ?
  `<button
    class="btn btn-red"
    data-action="reject"
    data-id="${b.id}">
    Reject
  </button>`
  :
  ""
}

</div>

</td>

</tr>

`;

    }).join("");
}

document.addEventListener("click",function(event){

  const button =
    event.target.closest("[data-action]");

  if(!button) return;

  const action =
    button.dataset.action;

  const id =
    button.dataset.id;

  if(action==="view"){
    openView(id);
  }

  if(action==="edit"){
    openEdit(id);
  }

  if(action==="payment"){
    markPaymentReceived(id);
  }

  if(action==="approve"){
    approveBusiness(id);
  }

  if(action==="reject"){
    rejectBusiness(id);
  }

});

async function markPaymentReceived(id){

  if(!confirm(
    "Confirm that payment has been received?"
  )) return;

  try{

    await supabaseRequest(
      `businesses?id=eq.${encodeURIComponent(id)}`,
      {
        method:"PATCH",
        headers:{
          "Prefer":"return=representation"
        },
        body:JSON.stringify({
          payment_status:"received"
        })
      }
    );

    showToast(
      "Payment marked as received."
    );

    await loadBusinesses();

  }catch(error){

    console.error(error);

    showToast(
      "Payment update failed: " + error.message
    );
  }
}

function escapeHtml(value){

  return String(value ?? "")
    .replace(/&/g,"&amp;")
    .replace(/</g,"&lt;")
    .replace(/>/g,"&gt;")
    .replace(/"/g,"&quot;")
    .replace(/'/g,"&#039;");
}

function escapeAttr(value){
  return escapeHtml(value);
}
